cmake_minimum_required(VERSION 3.16)
project(ps1_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ps1_core src/core.cpp src/gpu.cpp src/gpu_render.cpp src/gpu_video.cpp src/gpu_lines.cpp src/dma.cpp src/exe.cpp src/boot_io.cpp src/spu.cpp src/timers.cpp src/cdrom.cpp src/disc_image.cpp src/iso9660.cpp src/disc_boot.cpp src/sio.cpp)
target_include_directories(ps1_core PUBLIC src)
add_executable(ps1_lab src/main.cpp)
target_link_libraries(ps1_lab PRIVATE ps1_core)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE ps1_core)
add_test(NAME core_tests COMMAND core_tests)
if(WIN32)
    add_executable(ps1_ui WIN32 src/winmain.cpp)
    target_link_libraries(ps1_ui PRIVATE ps1_core user32 gdi32 comdlg32)
    target_compile_definitions(ps1_ui PRIVATE UNICODE _UNICODE NOMINMAX)
endif()
